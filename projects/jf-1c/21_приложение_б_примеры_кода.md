### Б.1 Пример сервиса с проверкой доступа

```java
@Service
@RequiredArgsConstructor
@Slf4j
public class TaskService {
    
    private final TaskRepository taskRepository;
    private final CrmAccessService crmAccessService;
    private final TaskMapper taskMapper;
    private final NotificationService notificationService;
    
    public TaskDto createTask(TaskCreateRequest request, Long userId) {
        // Проверка доступа к клиенту
        if (request.getClientId() != null) {
            if (!crmAccessService.canAccessClient(userId, request.getClientId())) {
                throw new AccessDeniedException("Cannot create task for this client");
            }
        }
        
        Task task = taskMapper.toEntity(request);
        task.setCreatedByUserId(userId);
        task.setStageId(request.getStageId());
        task.setStatus(TaskStatus.OPEN);
        task.setCreatedAt(LocalDateTime.now());
        task.setUpdatedAt(LocalDateTime.now());
        
        Task saved = taskRepository.save(task);
        
        log.info("Task created: id={}, by user={}", saved.getId(), userId);
        
        return taskMapper.toDto(saved);
    }
    
    public TaskDto updateTask(Long taskId, TaskUpdateRequest request, Long userId) {
        Task task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("Task not found: " + taskId));
        
        // Row-level access check
        if (!crmAccessService.canAccessTask(userId, task)) {
            throw new AccessDeniedException("Cannot modify this task");
        }
        
        // Business rule: can't modify completed tasks
        if (task.getStatus() == TaskStatus.DONE) {
            throw new BusinessException("Cannot modify completed task");
        }
        
        // Update fields
        if (request.getTitle() != null) {
            task.setTitle(request.getTitle());
        }
        if (request.getDescription() != null) {
            task.setDescription(request.getDescription());
        }
        if (request.getPriority() != null) {
            task.setPriority(request.getPriority());
        }
        if (request.getDueDate() != null) {
            task.setDueDate(request.getDueDate());
        }
        
        task.setUpdatedAt(LocalDateTime.now());
        Task saved = taskRepository.save(task);
        
        return taskMapper.toDto(saved);
    }
    
    public List<TaskDto> getTasksByStage(Long stageId, Long userId, String role) {
        // Different access rules for different roles
        if (role.equals("ADMIN") || role.equals("ADVISOR")) {
            // Full access
            return taskRepository.findByStageIdOrderByPosition(stageId)
                .stream()
                .map(taskMapper::toDto)
                .toList();
        } else {
            // Only assigned tasks
            return taskRepository.findByStageIdAndAssignedUserId(stageId, userId)
                .stream()
                .map(taskMapper::toDto)
                .toList();
        }
    }
    
    @Transactional
    public TaskDto assignTask(Long taskId, Long assigneeId, Long userId) {
        Task task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("Task not found"));
        
        // Only admin/advisor can reassign
        String userRole = crmAccessService.getUserRole(userId);
        if (!userRole.equals("ADMIN") && !userRole.equals("ADVISOR")) {
            if (!task.getAssignedUserId().equals(userId)) {
                throw new AccessDeniedException("Can only reassign own tasks");
            }
        }
        
        // Task Pool reopening logic
        if (task.getStage() != null && 
            task.getStage().getType() == StageType.LOST) {
            Long firstOpenStageId = pipelineService.getFirstOpenStageId(
                task.getStage().getPipelineId()
            );
            task.setStageId(firstOpenStageId);
            log.info("Task {} reopened from LOST to first OPEN stage", taskId);
        }
        
        task.setAssignedUserId(assigneeId);
        task.setUpdatedAt(LocalDateTime.now());
        
        Task saved = taskRepository.save(task);
        
        // Notify assignee
        notificationService.notifyTaskAssigned(assigneeId, saved);
        
        return taskMapper.toDto(saved);
    }
}
```

### Б.2 Пример контроллера с валидацией

```java
@RestController
@RequestMapping("/api/v1/tasks")
@RequiredArgsConstructor
@Slf4j
public class TaskController {
    
    private final TaskService taskService;
    private final UserPrincipal currentUser;
    
    @GetMapping
    public ResponseEntity<ApiResponse<List<TaskDto>>> getTasks(
            @RequestParam(required = false) Long stageId,
            @RequestParam(required = false) Long clientId,
            @RequestParam(required = false) Long assignedUserId,
            @RequestParam(required = false) TaskStatus status,
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(defaultValue = "20") int size,
            @RequestParam(defaultValue = "createdAt,desc") String sort) {
        
        TaskFilter filter = TaskFilter.builder()
            .stageId(stageId)
            .clientId(clientId)
            .assignedUserId(assignedUserId)
            .status(status)
            .page(page)
            .size(size)
            .sort(sort)
            .build();
        
        Page<TaskDto> result = taskService.getTasks(filter, currentUser.getId());
        
        return ResponseEntity.ok(ApiResponse.success(
            result.getContent(),
            PaginationMeta.fromPage(result)
        ));
    }
    
    @GetMapping("/{id}")
    public ResponseEntity<ApiResponse<TaskDto>> getTask(@PathVariable Long id) {
        TaskDto task = taskService.getTaskById(id, currentUser.getId());
        return ResponseEntity.ok(ApiResponse.success(task));
    }
    
    @PostMapping
    public ResponseEntity<ApiResponse<TaskDto>> createTask(
            @Valid @RequestBody TaskCreateRequest request) {
        
        TaskDto task = taskService.createTask(request, currentUser.getId());
        
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success(task));
    }
    
    @PutMapping("/{id}")
    public ResponseEntity<ApiResponse<TaskDto>> updateTask(
            @PathVariable Long id,
            @Valid @RequestBody TaskUpdateRequest request) {
        
        TaskDto task = taskService.updateTask(id, request, currentUser.getId());
        return ResponseEntity.ok(ApiResponse.success(task));
    }
    
    @DeleteMapping("/{id}")
    public ResponseEntity<ApiResponse<Void>> deleteTask(@PathVariable Long id) {
        taskService.deleteTask(id, currentUser.getId());
        return ResponseEntity.ok(ApiResponse.success(null));
    }
    
    @PostMapping("/{id}/subtasks")
    public ResponseEntity<ApiResponse<SubtaskDto>> addSubtask(
            @PathVariable Long id,
            @Valid @RequestBody SubtaskCreateRequest request) {
        
        SubtaskDto subtask = taskService.addSubtask(id, request, currentUser.getId());
        
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success(subtask));
    }
    
    @PostMapping("/{id}/comments")
    public ResponseEntity<ApiResponse<TaskCommentDto>> addComment(
            @PathVariable Long id,
            @Valid @RequestBody TaskCommentRequest request) {
        
        TaskCommentDto comment = taskService.addComment(id, request, currentUser.getId());
        
        return ResponseEntity
            .status(HttpStatus.CREATED)
            .body(ApiResponse.success(comment));
    }
    
    @PostMapping("/batch/assign")
    public ResponseEntity<ApiResponse<BatchOperationResult>> batchAssign(
            @Valid @RequestBody TaskBatchAssignRequest request) {
        
        BatchOperationResult result = taskService.batchAssign(
            request.getTaskIds(),
            request.getAssigneeId(),
            currentUser.getId()
        );
        
        return ResponseEntity.ok(ApiResponse.success(result));
    }
}
```

```typescript
// entities/task/api.ts
export const taskApi = {
  getTasks: (filter: TaskFilter) =>
    apiRequest.get<TaskListResponse>('/tasks', { params: filter }),
  
  getTask: (id: number) =>
    apiRequest.get<TaskDetailResponse>(`/tasks/${id}`),
  
  createTask: (data: TaskCreateData) =>
    apiRequest.post<TaskDetailResponse>('/tasks', data),
  
  updateTask: (id: number, data: TaskUpdateData) =>
    apiRequest.put<TaskDetailResponse>(`/tasks/${id}`, data),
  
  deleteTask: (id: number) =>
    apiRequest.delete<void>(`/tasks/${id}`),
  
  assignTask: (taskId: number, assigneeId: number) =>
    apiRequest.post<TaskDetailResponse>(`/tasks/${taskId}/assign`, { assigneeId }),
  
  addComment: (taskId: number, content: string) =>
    apiRequest.post<TaskCommentResponse>(`/tasks/${taskId}/comments`, { content }),
  
  batchAssign: (taskIds: number[], assigneeId: number) =>
    apiRequest.post<BatchAssignResponse>('/tasks/batch/assign', { taskIds, assigneeId }),
};

// entities/task/hooks/useTaskList.ts
export function useTaskList(filter: TaskFilter) {
  return useQuery({
    queryKey: ['tasks', 'list', filter],
    queryFn: () => taskApi.getTasks(filter),
    staleTime: 5 * 60 * 1000, // 5 minutes
    select: (data) => data.data,
  });
}

// entities/task/hooks/useTask.ts
export function useTask(taskId: number | null) {
  return useQuery({
    queryKey: ['tasks', 'detail', taskId],
    queryFn: () => taskApi.getTask(taskId!),
    enabled: !!taskId,
    select: (data) => data.data,
  });
}

// entities/task/hooks/useTaskMutations.ts
export function useCreateTask() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (data: TaskCreateData) => taskApi.createTask(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
      toast.success('Задача создана');
    },
    onError: (error: ApiError) => {
      toast.error(error.message);
    },
  });
}

export function useUpdateTask() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: ({ id, data }: { id: number; data: TaskUpdateData }) =>
      taskApi.updateTask(id, data),
    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
      queryClient.invalidateQueries({ queryKey: ['tasks', 'detail', variables.id] });
      toast.success('Задача обновлена');
    },
  });
}

export function useDeleteTask() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (id: number) => taskApi.deleteTask(id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
      toast.success('Задача удалена');
    },
  });
}

export function useAssignTask() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: ({ taskId, assigneeId }: { taskId: number; assigneeId: number }) =>
      taskApi.assignTask(taskId, assigneeId),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
      toast.success('Задача назначена');
    },
  });
}
```

### Б.4 Пример WebSocket Chat хука

```typescript
// features/chat/hooks/useChat.ts
export function useChat() {
  const { user } = useAuth();
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [contacts, setContacts] = useState<ChatContact[]>([]);
  const [activeContact, setActiveContact] = useState<ChatContact | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  
  const stompClient = useRef<Stomp.Client | null>(null);
  
  // Load contacts on mount
  useEffect(() => {
    chatApi.getContacts().then((response) => {
      setContacts(response.data);
    });
  }, []);
  
  // Load messages when active contact changes
  useEffect(() => {
    if (!activeContact) return;
    
    chatApi.getMessages(activeContact.id).then((response) => {
      setMessages(response.data);
    });
  }, [activeContact]);
  
  // Connect to WebSocket
  useEffect(() => {
    if (!user) return;
    
    const socket = new SockJS(getWsEndpointUrl());
    stompClient.current = Stomp.over(socket);
    
    stompClient.current.connect(
      { Authorization: `Bearer ${getAccessToken()}` },
      () => {
        setIsConnected(true);
        
        // Subscribe to personal messages
        stompClient.current?.subscribe(
          `/topic/chat/${user.id}`,
          (message) => {
            const newMessage = JSON.parse(message.body) as ChatMessage;
            setMessages((prev) => [...prev, newMessage]);
          },
          { Authorization: `Bearer ${getAccessToken()}` }
        );
      },
      (error) => {
        console.error('WebSocket connection error:', error);
        setIsConnected(false);
      }
    );
    
    return () => {
      if (stompClient.current) {
        stompClient.current.disconnect();
      }
    };
  }, [user]);
  
  const sendMessage = (content: string) => {
    if (!stompClient.current || !activeContact) return;
    
    const message: SendMessageRequest = {
      recipientId: activeContact.id,
      content,
      messageType: 'TEXT',
    };
    
    stompClient.current.send(
      '/app/chat.send',
      { Authorization: `Bearer ${getAccessToken()}` },
      JSON.stringify(message)
    );
    
    // Optimistic update
    const optimisticMessage: ChatMessage = {
      id: Date.now(), // temporary ID
      senderId: user!.id,
      recipientId: activeContact.id,
      content,
      messageType: 'TEXT',
      createdAt: new Date().toISOString(),
    };
    
    setMessages((prev) => [...prev, optimisticMessage]);
  };
  
  return {
    messages,
    contacts,
    activeContact,
    setActiveContact,
    sendMessage,
    isConnected,
  };
}
```
